Boundary effect simulation¶

This notebook simulates the boundary effect in dryland shrub-grass ecosystems—the phenomenon where vegetation at the desert margin exhibits enlarged patches and nonlinear decline under disturbance.

Code author: Jingyao Sun

The model captures two key mechanisms: positive feedback (clonal propagation and facilitation) maintains spatial coherence, while increasing disturbance progressively destroys it. Near a critical threshold, vegetation cover follows a power-law scaling $C \sim (P_c - P)^\beta$, the hallmark of a disorder-driven continuous (second-order) phase transition. The simulation uses species-specific interaction kernels (competition + facilitation), clonal propagation, and disturbance gradients to quantify this transition and its critical exponent.

# kernel generation tool
def create_circle_matrix(R, kernel_size):
    """
    Generate a circular binary kernel matrix.
    
    Parameters
    ----------
    R : int
        Radius of the circle (number of cells from center).
    kernel_size : int
        Size of the square kernel (must be odd).
    
    Returns
    -------
    matrix : 2D numpy array
        Binary matrix with 1s inside the circle and 0s outside.
    """
    if kernel_size % 2 == 0:
        raise ValueError("kernel_size must be an odd number")
    
    matrix = np.zeros((kernel_size, kernel_size), dtype=int)
    center_x, center_y = kernel_size // 2, kernel_size // 2
    
    for i in range(kernel_size):
        for j in range(kernel_size):
            distance = np.sqrt((i - center_x)**2 + (j - center_y)**2)
            if distance <= R:
                matrix[i, j] = 1
    return matrix

np.random.seed(43)  # Reproducibility
n = 200              # Lattice size (n x n cells)
resource = 0.7       # Initial resource supply level

# =============================================================================
# Species 1: Shrub kernel (Extensive strategy)
# =============================================================================
# Competition: wide-ranging but weak
kn1 = create_circle_matrix(40, 81)
nn1 = np.count_nonzero(kn1)
# Facilitation: small positive influence
kp1 = create_circle_matrix(4, 81)
np1 = np.count_nonzero(kp1)
# Combined kernel: competition (scaled by resource demand) - facilitation
kn1 = kn1 * (5 / resource) / nn1 - kp1 * 0.05

# =============================================================================
# Species 2: Grass kernel (Intensive strategy)
# =============================================================================
# Competition: spatially concentrated but strong
kn2 = create_circle_matrix(3, 7)
nn2 = np.count_nonzero(kn2)
kp2 = create_circle_matrix(1, 7)
np2 = np.count_nonzero(kp2)
kn2 = kn2 * (3 / resource) / nn2 - kp2 * 0.01

# =============================================================================

p2 = 0.0    # Grass initial density

steps = 2000

for i in range(steps):
    # ----- Dispersal -----
    # Offspring production probability proportional to current density
    p1 = 0.5 * np.count_nonzero(matrix1) / (n * n)
    p2 = 0.5 * np.count_nonzero(matrix2) / (n * n)
    
    # Multinomial draw for colonization of empty cells
    matrix_dispersal = np.random.choice(
        [0, 1, 2], size=(n, n),
        p=[1-p1-p2, p1, p2]
    ) * (1 - np.logical_or(matrix1, matrix2))
    
    matrix_dispersal1 = (matrix_dispersal == 1)
    matrix_dispersal2 = (matrix_dispersal == 2)
    
    # Combine existing and new recruits
    matrix_arena1 = np.logical_or(matrix_dispersal1, matrix1)
    matrix_arena2 = np.logical_or(matrix_dispersal2, matrix2)
    
    # ----- Interaction field -----
    # Convolve species distributions with their kernels
    pad_width1 = ((kn1.shape[0] - 1) // 2, (kn1.shape[1] - 1) // 2)
    matrix_ext1 = np.pad(matrix_arena1, pad_width=pad_width1, mode='symmetric')
    occupation1 = fftconvolve(matrix_ext1, kn1, mode='valid')
    
    pad_width2 = ((kn2.shape[0] - 1) // 2, (kn2.shape[1] - 1) // 2)
    matrix_ext2 = np.pad(matrix_arena2, pad_width=pad_width2, mode='symmetric')
    occupation2 = fftconvolve(matrix_ext2, kn2, mode='valid')
    
    # Net interaction field
    occupation = occupation1 + occupation2
    
    # ----- Survival probability and update -----
    # Species-specific survival thresholds
    prob1 = matrix1 - (occupation - 1.0) * 0.5
    prob2 = matrix2 - (occupation - 1.1) * 0.5
    
    matrix1 = (np.random.rand(n, n) < prob1) * matrix_arena1
    matrix2 = (np.random.rand(n, n) < prob2) * matrix_arena2
    
    density1.append(np.count_nonzero(matrix1) / (n * n))
    density2.append(np.count_nonzero(matrix2) / (n * n))
    
    if (i + 1) % 200 == 0:
        print(f"Iteration {i+1}/{steps}")

from scipy.optimize import curve_fit

# Convert to array
final1 = np.array(final1)
tolerance = 1e-12

# Find the critical threshold where cover drops to zero
idx_zero = np.where(final1 < tolerance)[0]
p_c = sequence[idx_zero[0]] if len(idx_zero) > 0 else sequence[-1]
print(f"Critical threshold p_c = {p_c:.4f}")

# Extract data below critical threshold
mask = (sequence < p_c) & (final1 > tolerance)
delta_p = (p_c - sequence[mask])
rho = final1[mask]

# Keep only near-critical data (within 10% of threshold)
mask_near = delta_p < 0.1
delta_p = delta_p[mask_near]
rho = rho[mask_near]

# ----- Power-law fitting -----
def powerlaw(x, A, beta):
    return A * x**beta

valid = (delta_p > 1e-6) & (rho > 0)
x = delta_p[valid]
y = rho[valid]

popt, pcov = curve_fit(powerlaw, x, y, p0=[0.3, 0.5])
A_fit, beta_fit = popt
A_err, beta_err = np.sqrt(np.diag(pcov))

# R² calculation
y_pred = powerlaw(x, *popt)
ss_res = np.sum((y - y_pred)**2)
ss_tot = np.sum((y - np.mean(y))**2)
r2 = 1 - ss_res / ss_tot

print(f"β = {beta_fit:.4f} ± {beta_err:.4f}")
print(f"A = {A_fit:.4e} ± {A_err:.4e}")
print(f"R² = {r2:.4f}")
